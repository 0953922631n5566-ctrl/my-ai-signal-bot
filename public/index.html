async function updateAutoScan() { try { const response = await fetch("/api/scan", { cache: "no-store" }); if (!response.ok) { throw new Error("Auto scan API unavailable"); } const data = await response.json(); if (!data.ok) return; // อัปเดตองค์ประกอบบนหน้าเว็บตาม ID const symbolEl = document.getElementById("selected-symbol"); const signalEl = document.getElementById("selected-signal"); const scoreEl = document.getElementById("selected-score"); if (symbolEl) { symbolEl.textContent = data.selected.symbol || "กำลังสแกน"; } if (signalEl) { signalEl.textContent = data.selected.signal; signalEl.dataset.signal = data.selected.signal; } if (scoreEl) { scoreEl.textContent = `${data.selected.score}%`; } // เก็บข้อมูลผลสแกนไว้ใช้งานต่อในหน้าเว็บ window.latestAutoScan = data; } catch (error) { console.error("Auto scan error:", error); } } // เรียกใช้งานทันทีและวนลูปทุกๆ 5 วินาที updateAutoScan(); setInterval(updateAutoScan, 5000);
